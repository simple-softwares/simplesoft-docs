## Scoring Formula (what the frontend displays)

```
points_attendance  = monthAttendance × 5
points_tasks       = tasksCompleted  × 10
points_early       = earlyCheckIns   × 3
points_streak      = streak ÷ 7 × 15   (15 pts per complete 7-day streak)
totalPoints        = sum of above
```

**monthAttendance** = count of distinct dates in `attendance_records` for current calendar month
where `user_id = current_user`.

**weekAttendance** = count of distinct dates for current ISO week (Mon–Sun).

**tasksCompleted** = count of tasks in `task_assignees` for this user where
`tasks.workspace_id = ws` and the linked `task_stage.is_closed = True`.

**earlyCheckIns** = count of `attendance_records` this month where
`EXTRACT(HOUR FROM check_in) < 9` (checked in before 09:00).

**streak** = longest run of consecutive calendar days ending on today or yesterday
that have at least one `attendance_record`.

**rank** = position of `current_user` when all workspace users are sorted by `totalPoints` DESC.

### Helper: compute stats for one user

```python
async def _compute_stats(user_id: int, ws_id: int, db: AsyncSession) -> dict:
    today = date.today()
    month_start = today.replace(day=1)
    week_start = today - timedelta(days=today.weekday())  # Monday

    # Month attendance — distinct dates
    m_r = await db.execute(
        select(func.count(distinct(AttendanceRecord.date)))
        .where(
            AttendanceRecord.user_id == user_id,
            AttendanceRecord.workspace_id == ws_id,
            AttendanceRecord.date >= month_start,
        )
    )
    month_attendance = m_r.scalar() or 0

    # Week attendance — distinct dates this week
    w_r = await db.execute(
        select(func.count(distinct(AttendanceRecord.date)))
        .where(
            AttendanceRecord.user_id == user_id,
            AttendanceRecord.workspace_id == ws_id,
            AttendanceRecord.date >= week_start,
        )
    )
    week_attendance = w_r.scalar() or 0

    # Early check-ins this month (before 09:00)
    e_r = await db.execute(
        select(func.count())
        .where(
            AttendanceRecord.user_id == user_id,
            AttendanceRecord.workspace_id == ws_id,
            AttendanceRecord.date >= month_start,
            extract('hour', AttendanceRecord.check_in) < 9,
        )
    )
    early_check_ins = e_r.scalar() or 0

    # Tasks completed (all-time) — join through task_assignees
    t_r = await db.execute(
        select(func.count(Task.id))
        .join(task_assignees, task_assignees.c.task_id == Task.id)
        .join(TaskStage, Task.stage_id == TaskStage.id)
        .where(
            task_assignees.c.user_id == user_id,
            Task.workspace_id == ws_id,
            TaskStage.is_closed == True,
        )
    )
    tasks_completed = t_r.scalar() or 0

    # Streak — load distinct attendance dates for last 60 days, compute run
    s_r = await db.execute(
        select(distinct(AttendanceRecord.date))
        .where(
            AttendanceRecord.user_id == user_id,
            AttendanceRecord.workspace_id == ws_id,
            AttendanceRecord.date >= today - timedelta(days=60),
        )
        .order_by(AttendanceRecord.date.desc())
    )
    dates = sorted({row[0] for row in s_r.all()}, reverse=True)

    streak = 0
    if dates:
        anchor = dates[0]
        # Allow streak to end on today or yesterday
        if anchor >= today - timedelta(days=1):
            streak = 1
            for i in range(1, len(dates)):
                if (dates[i - 1] - dates[i]).days == 1:
                    streak += 1
                else:
                    break

    # Points
    pts_attendance = month_attendance * 5
    pts_tasks      = tasks_completed * 10
    pts_early      = early_check_ins * 3
    pts_streak     = (streak // 7) * 15
    total          = pts_attendance + pts_tasks + pts_early + pts_streak

    return {
        "monthAttendance": month_attendance,
        "weekAttendance":  week_attendance,
        "tasksCompleted":  tasks_completed,
        "earlyCheckIns":   early_check_ins,
        "streak":          streak,
        "totalPoints":     total,
        "breakdown": {
            "attendance":  pts_attendance,
            "tasks":       pts_tasks,
            "earlyBonus":  pts_early,
            "streakBonus": pts_streak,
        },
    }
```

```python
async def _compute_badges(user_id: int, ws_id: int, stats: dict, db: AsyncSession) -> list:
    # Badge 1 needs all-time attendance count
    alltime_r = await db.execute(
        select(func.count()).where(
            AttendanceRecord.user_id == user_id,
            AttendanceRecord.workspace_id == ws_id,
        )
    )
    alltime_att = alltime_r.scalar() or 0

    # Badge 7 needs all-time tasks completed
    alltime_t_r = await db.execute(
        select(func.count(Task.id))
        .join(task_assignees, task_assignees.c.task_id == Task.id)
        .join(TaskStage, Task.stage_id == TaskStage.id)
        .where(
            task_assignees.c.user_id == user_id,
            Task.workspace_id == ws_id,
            TaskStage.is_closed == True,
        )
    )
    alltime_tasks = alltime_t_r.scalar() or 0

    criteria = {
        1: alltime_att > 0,
        2: stats["monthAttendance"] >= 5,
        3: alltime_tasks >= 10,
        4: stats["earlyCheckIns"] >= 3,
        5: stats["streak"] >= 7,
        6: stats["monthAttendance"] >= 15,
        7: alltime_tasks >= 25,
        8: stats["monthAttendance"] >= 22,
    }

    return [
        {**b, "earned": criteria.get(b["id"], False), "count": 1 if criteria.get(b["id"], False) else 0}
        for b in BADGE_CATALOG
    ]
```
