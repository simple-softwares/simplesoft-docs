## Why This Session

`AutomationPage.jsx` is a fully-built workflow editor with node picker, trigger
configuration, run history, and activate/deactivate toggles. All 6 `/automation/*`
stubs return empty data.

This is also the final stub-clearing session. After this, `stubs.py` will only retain

(`/{product_id}`, `/{product_id}/stock`, `/categories`).

**Execution engine:** Not needed. The `run` endpoint creates a dummy run record with
`status='success'` and synthetic steps derived from the workflow's nodes. This gives
the Runs tab real data without a scheduler or actual node execution.

**Important:** The frontend calls `api.put(...)` (not patch) when saving an existing
workflow. The stubs defined PATCH — the real router must register PUT.

import datetime, json, uuid
from sqlalchemy import ForeignKey, String, Text, Boolean, Integer, func
from sqlalchemy.orm import Mapped, mapped_column
from ..database import Base


class AutomationWorkflow(Base):
    __tablename__ = "automation_workflows"

    id:                   Mapped[int]  = mapped_column(primary_key=True)
    workspace_id:         Mapped[int]  = mapped_column(ForeignKey("workspaces.id"), index=True)
    name:                 Mapped[str]  = mapped_column(String(256))
    description:          Mapped[str]  = mapped_column(Text, default="")
    trigger_type:         Mapped[str]  = mapped_column(String(32), default="manual")
    trigger_model:        Mapped[str]  = mapped_column(String(128), default="")
    trigger_field:        Mapped[str]  = mapped_column(String(128), default="")
    cron_interval_number: Mapped[int]  = mapped_column(Integer, default=1)
    cron_interval_type:   Mapped[str]  = mapped_column(String(16), default="days")
    nodes:                Mapped[str]  = mapped_column(Text, default="[]")   # JSON
    edges:                Mapped[str]  = mapped_column(Text, default="[]")   # JSON
    active:               Mapped[bool] = mapped_column(Boolean, default=False)
    webhook_token:        Mapped[str]  = mapped_column(String(64), default="")
    run_count:            Mapped[int]  = mapped_column(Integer, default=0)
    success_count:        Mapped[int]  = mapped_column(Integer, default=0)
    last_run_at:          Mapped[datetime.datetime | None] = mapped_column(nullable=True)
    created_at:           Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now()
    )


class AutomationRun(Base):
    __tablename__ = "automation_runs"

    id:          Mapped[int] = mapped_column(primary_key=True)
    workflow_id: Mapped[int] = mapped_column(ForeignKey("automation_workflows.id"), index=True)

    status:      Mapped[str] = mapped_column(String(16), default="success")
    error:       Mapped[str] = mapped_column(Text, default="")
    steps:       Mapped[str] = mapped_column(Text, default="[]")   # JSON
    duration_ms: Mapped[int] = mapped_column(Integer, default=0)
    created_at:  Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now()

#### `POST /automation/workflows`
```python
@router.post("/automation/workflows", status_code=201)
async def create_workflow(
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    trigger = body.get("trigger_type", "manual")
    wf = AutomationWorkflow(
        workspace_id=ws_id,
        name=body.get("name") or "New Workflow",
        description=body.get("description") or "",
        trigger_type=trigger,
        trigger_model=body.get("trigger_model") or "",
        trigger_field=body.get("trigger_field") or "",
        cron_interval_number=body.get("cron_interval_number") or 1,
        cron_interval_type=body.get("cron_interval_type") or "days",
        nodes=json.dumps(body.get("nodes") or []),
        edges=json.dumps(body.get("edges") or []),
        active=False,
        webhook_token=str(uuid.uuid4()) if trigger == "webhook" else "",
    )
    db.add(wf)
    await db.commit()
    await db.refresh(wf)
    return _wf_out(wf)
```

#### `PUT /automation/workflows/{workflow_id}`
```python
@router.put("/automation/workflows/{workflow_id}")
async def update_workflow(
    workflow_id: int,
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(AutomationWorkflow).where(
            AutomationWorkflow.id == workflow_id,
            AutomationWorkflow.workspace_id == ws_id,
        )
    )
    wf = r.scalar_one_or_none()
    if not wf:
        raise HTTPException(404)
    for field in ("name", "description", "trigger_model", "trigger_field",
                  "cron_interval_number", "cron_interval_type"):
        if field in body:
            setattr(wf, field, body[field])
    if "trigger_type" in body:
        wf.trigger_type = body["trigger_type"]
        if body["trigger_type"] == "webhook" and not wf.webhook_token:
            wf.webhook_token = str(uuid.uuid4())
    if "nodes" in body:
        wf.nodes = json.dumps(body["nodes"])
    if "edges" in body:
        wf.edges = json.dumps(body["edges"])
    await db.commit()
    await db.refresh(wf)
    return _wf_out(wf)
```

#### `POST /automation/workflows/{workflow_id}/run`

Creates a synthetic run record with a step per node. Updates run stats on the workflow.

```python
@router.post("/automation/workflows/{workflow_id}/run")
async def run_workflow(
    workflow_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(AutomationWorkflow).where(
            AutomationWorkflow.id == workflow_id,
            AutomationWorkflow.workspace_id == ws_id,
        )
    )
    wf = r.scalar_one_or_none()
    if not wf:
        raise HTTPException(404)

    nodes = json.loads(wf.nodes or "[]")
    steps = [
        {
            "node_id":    n.get("id", f"node_{i}"),
            "node_type":  n.get("type", "unknown"),
            "node_label": n.get("label") or n.get("type", "Step"),
            "status":     "success",
            "error":      None,
        }
        for i, n in enumerate(nodes)
    ]

    run = AutomationRun(
        workflow_id=workflow_id,
        workspace_id=ws_id,
        status="success",
        error="",
        steps=json.dumps(steps),
        duration_ms=len(nodes) * 12,
    )
    db.add(run)

    wf.run_count     += 1
    wf.success_count += 1
    wf.last_run_at    = datetime.datetime.utcnow()
    await db.commit()
    return {"status": "triggered"}
```
