## Data Model — `models/sales.py`

```python
from sqlalchemy import Column, Integer, String, Numeric, DateTime, ForeignKey, func
from sqlalchemy.orm import Mapped, mapped_column
from ..database import Base
import datetime

class SalesOrder(Base):
    __tablename__ = "sales_orders"

    id:            Mapped[int]   = mapped_column(primary_key=True)
    workspace_id:  Mapped[int]   = mapped_column(ForeignKey("workspaces.id"), index=True)
    name:          Mapped[str]   = mapped_column(String(120), default="")
    partner_name:  Mapped[str]   = mapped_column(String(255), default="")
    partner_email: Mapped[str]   = mapped_column(String(255), default="")
    partner_phone: Mapped[str]   = mapped_column(String(60), default="")
    state:         Mapped[str]   = mapped_column(String(20), default="draft")
    amount_total:  Mapped[float] = mapped_column(Numeric(12, 2), default=0)
    user_id:       Mapped[int | None] = mapped_column(ForeignKey("users.id"), nullable=True)
    date_order:    Mapped[datetime.datetime | None] = mapped_column(nullable=True)
    note:          Mapped[str]   = mapped_column(String(2000), default="")
    created_at:    Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now()
    )
```

**Valid states:** `draft` → `sent` → `sale` (confirmed) → `done` (locked); `cancel` at any non-done state.

### Endpoints

#### `GET /sales/orders`

```python
@router.get("/sales/orders")
async def list_orders(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = get_workspace_id(current_user)
    result = await db.execute(
        select(SalesOrder)
        .where(SalesOrder.workspace_id == ws_id)
        .order_by(SalesOrder.created_at.desc())
    )
    orders = result.scalars().all()
    # Load user names in one query
    user_ids = {o.user_id for o in orders if o.user_id}
    users = {}
    if user_ids:
        ur = await db.execute(select(User).where(User.id.in_(user_ids)))
        users = {u.id: u for u in ur.scalars().all()}
    return [_order_out(o, users.get(o.user_id)) for o in orders]
```

#### `GET /sales/customers`

Aggregates distinct `partner_name` values from orders in this workspace, summing
`amount_total` across confirmed+done orders for `total_spent`.

```python
@router.get("/sales/customers")
async def list_customers(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = get_workspace_id(current_user)
    # Group by partner_name; sum amount_total for sale/done states
    result = await db.execute(
        select(
            SalesOrder.partner_name,
            SalesOrder.partner_email,
            SalesOrder.partner_phone,
            func.coalesce(
                func.sum(
                    func.case(
                        (SalesOrder.state.in_(["sale", "done"]), SalesOrder.amount_total),
                        else_=0,
                    )
                ),
                0,
            ).label("total_spent"),
            func.count(SalesOrder.id).label("order_count"),
        )
        .where(
            SalesOrder.workspace_id == ws_id,
            SalesOrder.partner_name != "",
        )
        .group_by(SalesOrder.partner_name, SalesOrder.partner_email, SalesOrder.partner_phone)
        .order_by(func.sum(SalesOrder.amount_total).desc())
    )
    rows = result.all()
    return [
        {
            "id":          i + 1,  # synthetic id for React key
            "name":        r.partner_name,
            "email":       r.partner_email or None,
            "phone":       r.partner_phone or None,
            "total_spent": float(r.total_spent),
            "order_count": r.order_count,
        }
        for i, r in enumerate(rows)
    ]
```

#### `POST /sales/orders`

```python
@router.post("/sales/orders", status_code=201)
async def create_order(
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),

        name=body.get("name", ""),
        partner_name=body.get("partner_name", ""),
        partner_email=body.get("partner_email", ""),
        partner_phone=body.get("partner_phone", ""),

        user_id=current_user.id,
        date_order=datetime.utcnow(),
        note=body.get("note", ""),
    )
    db.add(order)
    await db.commit()
    await db.refresh(order)
    return _order_out(order, current_user)
```

#### `GET /sales/orders/{order_id}`

```python
@router.get("/sales/orders/{order_id}")
async def get_order(
    order_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = get_workspace_id(current_user)
    r = await db.execute(
        select(SalesOrder).where(SalesOrder.id == order_id, SalesOrder.workspace_id == ws_id)
    )
    order = r.scalar_one_or_none()
    if not order:
        raise HTTPException(404)
    user = None
    if order.user_id:
        ur = await db.execute(select(User).where(User.id == order.user_id))
        user = ur.scalar_one_or_none()
    return _order_out(order, user)
```

#### `PATCH /sales/orders/{order_id}`

```python
@router.patch("/sales/orders/{order_id}")
async def update_order(
    order_id: int,
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = get_workspace_id(current_user)
    r = await db.execute(
        select(SalesOrder).where(SalesOrder.id == order_id, SalesOrder.workspace_id == ws_id)
    )
    order = r.scalar_one_or_none()
    if not order:
        raise HTTPException(404)
    if order.state == "done":
        raise HTTPException(400, "Locked orders cannot be edited")
    for field in ("name", "partner_name", "partner_email", "partner_phone", "note", "amount_total"):
        if field in body:
            setattr(order, field, body[field])
    if "state" in body and body["state"] in ("draft", "sent", "sale", "done", "cancel"):
        order.state = body["state"]
    await db.commit()
    await db.refresh(order)
    return _order_out(order)
```

#### `POST /sales/orders/{order_id}/confirm`

```python
@router.post("/sales/orders/{order_id}/confirm")
async def confirm_order(
    order_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = get_workspace_id(current_user)
    r = await db.execute(
        select(SalesOrder).where(SalesOrder.id == order_id, SalesOrder.workspace_id == ws_id)
    )
    order = r.scalar_one_or_none()
    if not order:
        raise HTTPException(404)
    if order.state in ("sale", "done"):
        return {"id": order.id, "state": order.state}
    order.state = "sale"
    await db.commit()
    return {"id": order.id, "state": "sale"}
```
