## Why This Session

`FilesPage.jsx` is a fully-built file manager with folder navigation, breadcrumbs,
grid/list toggle, drag-and-drop upload, image/PDF preview, and a download button.
All 9 `/files/*` stubs return empty data so the page shows nothing.

**Storage decision:** File content stored as `BYTEA` in PostgreSQL. No filesystem
or S3 configuration needed — runs anywhere without extra setup.

**Auth decision:** The download endpoint requires NO authentication. The frontend
uses `/api/files/${file.id}/download` directly as `<img src>` and `<iframe src>` for
preview, and browsers cannot send custom headers to those URLs. Without public access
the preview modal would show broken images.

## What the Frontend Expects

### How navigation works
- Root (`currentFolder.id = null`): loads `GET /files/directories?parent_id=null` + `GET /files/recent`
- Inside folder: loads `GET /files/directories?parent_id=X` + `GET /files?directory_id=X`

### Directory shape
```json
{ "id": 1, "name": "Documents", "count_files": 3, "count_directories": 1 }
```

### File shape
```json
{
  "id": 1,
  "name": "report.pdf",
  "mimetype": "application/pdf",
  "file_size": 204800,
  "create_date": "2026-05-21T10:30:00",
  "write_date": "2026-05-21T10:30:00"
}
```
**Note:** NO `content` field in list/detail responses — only the download endpoint streams bytes.

### API calls
| Stub | Purpose |
|------|---------|
| `GET /files/directories?parent_id=X` | List subdirectories of X (null = root) |
| `POST /files/directories` | Create folder: `{ name, parent_id }` |
| `DELETE /files/directories/{id}` | Delete folder + contents |
| `GET /files/recent` | Last 20 files workspace-wide (root view) |
| `GET /files?directory_id=X` | Files in folder X |
| `POST /files/upload` | Multipart: `file` + `directory_id` form fields |
| `GET /files/{id}/download` | Stream file bytes — **NO AUTH** |
| `PATCH /files/{id}` | Rename: `{ name }` |
| `DELETE /files/{id}` | Delete file |

```python
import datetime
from sqlalchemy import ForeignKey, String, LargeBinary, Integer, func
from sqlalchemy.orm import Mapped, mapped_column
from ..database import Base


class FileDirectory(Base):
    __tablename__ = "file_directories"

    id:           Mapped[int] = mapped_column(primary_key=True)
    workspace_id: Mapped[int] = mapped_column(ForeignKey("workspaces.id"), index=True)
    parent_id:    Mapped[int | None] = mapped_column(ForeignKey("file_directories.id"), nullable=True, index=True)
    name:         Mapped[str] = mapped_column(String(255))
    created_by:   Mapped[int | None] = mapped_column(ForeignKey("users.id"), nullable=True)
    created_at:   Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now()
    )


class FileRecord(Base):
    __tablename__ = "file_records"

    id:           Mapped[int]   = mapped_column(primary_key=True)
    workspace_id: Mapped[int]   = mapped_column(ForeignKey("workspaces.id"), index=True)
    directory_id: Mapped[int | None] = mapped_column(ForeignKey("file_directories.id"), nullable=True, index=True)
    name:         Mapped[str]   = mapped_column(String(512))
    mimetype:     Mapped[str]   = mapped_column(String(128), default="application/octet-stream")
    file_size:    Mapped[int]   = mapped_column(Integer, default=0)
    content:      Mapped[bytes] = mapped_column(LargeBinary)
    created_by:   Mapped[int | None] = mapped_column(ForeignKey("users.id"), nullable=True)
    created_at:   Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now()
    )
    updated_at:   Mapped[datetime.datetime] = mapped_column(
        default=datetime.datetime.utcnow, server_default=func.now(), onupdate=datetime.datetime.utcnow
    )
```

### Helper

```python
def _dir_out(d: FileDirectory, count_files: int = 0, count_dirs: int = 0) -> dict:
    return {
        "id":                d.id,
        "name":              d.name,
        "count_files":       count_files,
        "count_directories": count_dirs,
    }

def _file_out(f: FileRecord) -> dict:
    return {
        "id":          f.id,
        "name":        f.name,
        "mimetype":    f.mimetype,
        "file_size":   f.file_size,
        "create_date": f.created_at.isoformat(),
        "write_date":  f.updated_at.isoformat(),
    }
```

### `GET /files/directories`

```python
@router.get("/files/directories")
async def list_directories(
    parent_id: Optional[int] = Query(None),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    dirs_r = await db.execute(
        select(FileDirectory).where(
            FileDirectory.workspace_id == ws_id,
            FileDirectory.parent_id == parent_id,
        ).order_by(FileDirectory.name.asc())
    )
    dirs = dirs_r.scalars().all()

    result = []
    for d in dirs:
        cf_r = await db.execute(
            select(func.count(FileRecord.id)).where(FileRecord.directory_id == d.id)
        )
        cd_r = await db.execute(
            select(func.count(FileDirectory.id)).where(FileDirectory.parent_id == d.id)
        )
        result.append(_dir_out(d, cf_r.scalar() or 0, cd_r.scalar() or 0))
    return result
```

### `POST /files/directories`

```python
@router.post("/files/directories", status_code=201)
async def create_directory(
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    name = (body.get("name") or "").strip()
    if not name:
        raise HTTPException(400, "Folder name required")
    parent_id = body.get("parent_id")
    d = FileDirectory(workspace_id=ws_id, parent_id=parent_id, name=name, created_by=current_user.id)
    db.add(d)
    await db.commit()
    await db.refresh(d)
    return _dir_out(d)
```

### `DELETE /files/directories/{directory_id}`

Deletes all files inside the folder first, then the folder itself.
(For nested folders, this only deletes direct children; deep nesting is rarely used in practice.)

```python
@router.delete("/files/directories/{directory_id}", status_code=204)
async def delete_directory(
    directory_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(FileDirectory).where(FileDirectory.id == directory_id, FileDirectory.workspace_id == ws_id)
    )
    d = r.scalar_one_or_none()
    if not d:
        raise HTTPException(404)
    # Delete all files inside
    await db.execute(sa_delete(FileRecord).where(FileRecord.directory_id == directory_id))
    await db.delete(d)
    await db.commit()
```

### `GET /files/recent`

```python
@router.get("/files/recent")
async def recent_files(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(FileRecord)
        .where(FileRecord.workspace_id == ws_id)
        .order_by(FileRecord.created_at.desc())
        .limit(20)
    )
    return [_file_out(f) for f in r.scalars().all()]
```

### `POST /files/upload`

```python
@router.post("/files/upload", status_code=201)
async def upload_file(
    file: UploadFile = File(...),
    directory_id: Optional[int] = Form(None),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    content = await file.read()
    rec = FileRecord(
        workspace_id=ws_id,
        directory_id=directory_id,
        name=file.filename or "upload",
        mimetype=file.content_type or "application/octet-stream",
        file_size=len(content),
        content=content,
        created_by=current_user.id,
    )
    db.add(rec)
    await db.commit()
    await db.refresh(rec)
    return _file_out(rec)
```

### `GET /files/{file_id}/download` — NO AUTH

```python
@router.get("/files/{file_id}/download")
async def download_file(
    file_id: int,
    db: AsyncSession = Depends(get_db),
):
    r = await db.execute(select(FileRecord).where(FileRecord.id == file_id))
    rec = r.scalar_one_or_none()
    if not rec:
        raise HTTPException(404)
    return Response(
        content=rec.content,
        media_type=rec.mimetype,
        headers={"Content-Disposition": f'inline; filename="{rec.name}"'},
    )
```

### `PATCH /files/{file_id}`

```python
@router.patch("/files/{file_id}")
async def rename_file(
    file_id: int,
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(FileRecord).where(FileRecord.id == file_id, FileRecord.workspace_id == ws_id)
    )
    rec = r.scalar_one_or_none()
    if not rec:
        raise HTTPException(404)
    if "name" in body:
        rec.name = body["name"]
    await db.commit()
    await db.refresh(rec)
    return _file_out(rec)
```

## Route ordering note

`GET /files/recent` and `GET /files` are both registered before `GET /files/{file_id}/download`.
FastAPI matches `/files/recent` as a literal before `{file_id}`, so ordering matters — register
`recent` route before the `{file_id}` parameterized route. Since they're different paths this
is fine, but keep `GET /files/recent` defined above `GET /files/{file_id}/download` in the file.
