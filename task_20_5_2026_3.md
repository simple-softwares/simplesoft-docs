from ..database import Base


class ChartOfAccount(Base):
    __tablename__ = "chart_of_accounts"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    workspace_id: Mapped[int] = mapped_column(ForeignKey("workspaces.id"), nullable=False, index=True)
    account_code: Mapped[str] = mapped_column(String(8), nullable=False)
    name: Mapped[str] = mapped_column(String(255), nullable=False)
    account_type: Mapped[str] = mapped_column(String(16), nullable=False)  # asset|liability|equity|income|expense
    is_system: Mapped[bool] = mapped_column(Boolean, default=False)        # system accounts cannot be deleted
    is_active: Mapped[bool] = mapped_column(Boolean, default=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)


class JournalEntry(Base):
    __tablename__ = "journal_entries"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    workspace_id: Mapped[int] = mapped_column(ForeignKey("workspaces.id"), nullable=False, index=True)
    entry_date: Mapped[date] = mapped_column(Date, nullable=False)
    reference: Mapped[Optional[str]] = mapped_column(String(64), nullable=True)   # e.g. INV-2026-00001
    description: Mapped[Optional[str]] = mapped_column(String(500), nullable=True)
    is_system: Mapped[bool] = mapped_column(Boolean, default=False)                # auto-created entries
    created_by: Mapped[Optional[int]] = mapped_column(ForeignKey("users.id"), nullable=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    lines: Mapped[list["JournalEntryLine"]] = relationship(
        "JournalEntryLine", cascade="all, delete-orphan", lazy="selectin"
    )


class JournalEntryLine(Base):
    __tablename__ = "journal_entry_lines"

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    entry_id: Mapped[int] = mapped_column(
        ForeignKey("journal_entries.id", ondelete="CASCADE"), nullable=False, index=True
    )
    account_id: Mapped[int] = mapped_column(ForeignKey("chart_of_accounts.id"), nullable=False)
    debit: Mapped[float] = mapped_column(Numeric(12, 2), default=0)
    credit: Mapped[float] = mapped_column(Numeric(12, 2), default=0)
    description: Mapped[Optional[str]] = mapped_column(String(255), nullable=True)
```

@router.post("/journal-entries", response_model=JournalEntryOut, status_code=201)
async def create_journal_entry(
    body: JournalEntryIn,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    total_debit = round(sum(l.debit for l in body.lines), 2)
    total_credit = round(sum(l.credit for l in body.lines), 2)
    if abs(total_debit - total_credit) > 0.01:
        raise HTTPException(400, f"Journal entry is not balanced: debits ₹{total_debit} ≠ credits ₹{total_credit}")
    if total_debit == 0:
        raise HTTPException(400, "Journal entry must have at least one non-zero amount.")

    entry = JournalEntry(
        workspace_id=current_user.workspace_id,
        entry_date=body.entry_date,
        reference=body.reference,
        description=body.description,
        is_system=False,
        created_by=current_user.id,
    )
    for li in body.lines:
        entry.lines.append(JournalEntryLine(
            account_id=li.account_id,
            debit=li.debit,
            credit=li.credit,
            description=li.description,
        ))
    db.add(entry)
    await db.commit()
    await db.refresh(entry)
    # Re-fetch with enriched lines via list endpoint logic
    return await _enrich_entry(entry, db)


async def _enrich_entry(entry: JournalEntry, db: AsyncSession) -> JournalEntryOut:
    lines_out = []
    for line in entry.lines:
        acc = await db.get(ChartOfAccount, line.account_id)

            id=line.id,
            account_id=line.account_id,
            account_code=acc.account_code if acc else None,
            account_name=acc.name if acc else None,
            debit=float(line.debit),
            credit=float(line.credit),
            description=line.description,
        ))
    return JournalEntryOut(
        id=entry.id,
        entry_date=entry.entry_date,
        reference=entry.reference,
        description=entry.description,
        is_system=entry.is_system,
        created_at=entry.created_at,

@router.get("/reports/trial-balance")
async def trial_balance(
    from_date: Optional[date] = Query(None),
    to_date: Optional[date] = Query(None),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    await seed_accounts_if_empty(current_user.workspace_id, db)

    accounts_result = await db.execute(
        select(ChartOfAccount).where(
            ChartOfAccount.workspace_id == current_user.workspace_id,
            ChartOfAccount.is_active == True,
        ).order_by(ChartOfAccount.account_code)
    )
    accounts = accounts_result.scalars().all()

    rows = []
    total_debit = 0.0
    total_credit = 0.0

    for acc in accounts:
        q = (
            select(
                func.coalesce(func.sum(JournalEntryLine.debit), 0).label("debit"),
                func.coalesce(func.sum(JournalEntryLine.credit), 0).label("credit"),
            )
            .join(JournalEntry, JournalEntry.id == JournalEntryLine.entry_id)
            .where(
                JournalEntryLine.account_id == acc.id,
                JournalEntry.workspace_id == current_user.workspace_id,
            )
        )
        if from_date:
            q = q.where(JournalEntry.entry_date >= from_date)
        if to_date:
            q = q.where(JournalEntry.entry_date <= to_date)

        r = await db.execute(q)
        row = r.one()
        debit = float(row.debit)
        credit = float(row.credit)
        if debit == 0 and credit == 0:
            continue

        balance = debit - credit
        total_debit += debit
        total_credit += credit
        rows.append({
            "account_code": acc.account_code,
            "account_name": acc.name,
            "account_type": acc.account_type,
            "debit": round(debit, 2),
            "credit": round(credit, 2),
            "balance": round(balance, 2),
        })

    return {
        "from_date": from_date,
        "to_date": to_date,
        "rows": rows,
        "total_debit": round(total_debit, 2),
        "total_credit": round(total_credit, 2),
    }

                    func.coalesce(func.sum(JournalEntryLine.debit), 0).label("debit"),

        return rows, round(section_total, 2)

    expense_rows, total_expenses = await section_totals("expense")
    net_profit = round(total_income - total_expenses, 2)

    return {
        "from_date": from_date,
        "to_date": to_date,
        "income": {"rows": income_rows, "total": total_income},
        "expenses": {"rows": expense_rows, "total": total_expenses},

@router.post("/{invoice_id}/send", response_model=InvoiceOut)
async def send_invoice(
    invoice_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(
        select(Invoice).where(

    invoice = result.scalar_one_or_none()

    if invoice.status != "draft":
        raise HTTPException(409, f"Invoice is already {invoice.status}.")
    if not invoice.lines:
        raise HTTPException(400, "Cannot send an invoice with no line items.")

    invoice.status = "sent"

    # Auto-create journal entry: DR Accounts Receivable / CR Revenue + GST Payable
    await seed_accounts_if_empty(current_user.workspace_id, db)
    ar_acc = await get_account_by_code(current_user.workspace_id, "1010", db)
    rev_acc = await get_account_by_code(current_user.workspace_id, "4001", db)
    cgst_acc = await get_account_by_code(current_user.workspace_id, "2010", db)
    sgst_acc = await get_account_by_code(current_user.workspace_id, "2011", db)
    igst_acc = await get_account_by_code(current_user.workspace_id, "2012", db)

    if ar_acc and rev_acc:
        taxable = round(float(invoice.subtotal) - float(invoice.discount_total), 2)
        entry = JournalEntry(
            workspace_id=current_user.workspace_id,
            entry_date=invoice.invoice_date,

        entry.lines.append(JournalEntryLine(account_id=ar_acc.id, debit=float(invoice.total), credit=0))
        entry.lines.append(JournalEntryLine(account_id=rev_acc.id, debit=0, credit=taxable))
        if float(invoice.cgst_total) > 0 and cgst_acc:
            entry.lines.append(JournalEntryLine(account_id=cgst_acc.id, debit=0, credit=float(invoice.cgst_total)))

        db.add(entry)

    await db.commit()
    await db.refresh(invoice)

@router.post("/{invoice_id}/mark-paid", response_model=InvoiceOut)
async def mark_paid(
    invoice_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(
        select(Invoice).where(
            Invoice.id == invoice_id,
            Invoice.workspace_id == current_user.workspace_id,
        )
    )
    invoice = result.scalar_one_or_none()
    if not invoice:
        raise HTTPException(404, "Invoice not found")
    if invoice.status not in ("sent", "overdue"):
        raise HTTPException(409, f"Cannot mark a '{invoice.status}' invoice as paid.")

    invoice.status = "paid"

    # Auto-create journal entry: DR Bank / CR Accounts Receivable
    await seed_accounts_if_empty(current_user.workspace_id, db)
    bank_acc = await get_account_by_code(current_user.workspace_id, "1002", db)
    ar_acc = await get_account_by_code(current_user.workspace_id, "1010", db)

    if bank_acc and ar_acc:
        entry = JournalEntry(
            workspace_id=current_user.workspace_id,
            entry_date=date.today(),
            reference=invoice.invoice_number,
            description=f"Payment received from {invoice.customer_name}",
            is_system=True,
        )
        entry.lines.append(JournalEntryLine(account_id=bank_acc.id, debit=float(invoice.total), credit=0))
        entry.lines.append(JournalEntryLine(account_id=ar_acc.id, debit=0, credit=float(invoice.total)))
        db.add(entry)

    await db.commit()
    await db.refresh(invoice)
    return invoice
```

Also add `from datetime import date` to the invoices.py imports if not already there.
