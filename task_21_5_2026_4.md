### Helper

def _event_out(ev: CalendarEvent) -> dict:
    return {
        "id":          ev.id,
        "name":        ev.name,
        "start":       ev.start.isoformat() if ev.start else None,
        "stop":        ev.stop.isoformat() if ev.stop else None,
        "location":    ev.location or None,
        "description": ev.description or None,
        "all_day":     ev.all_day,
    }

### Endpoints

#### `GET /calendar/events`

Supports optional `date_from` and `date_to` query params (YYYY-MM-DD strings).
If provided, filters events whose `start` falls within the range (inclusive).
Without params, returns all workspace events ordered by start ASC.

```python
@router.get("/calendar/events")
async def list_events(
    date_from: Optional[str] = Query(None),
    date_to:   Optional[str] = Query(None),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    q = select(CalendarEvent).where(CalendarEvent.workspace_id == ws_id)
    if date_from:
        q = q.where(CalendarEvent.start >= datetime.datetime.fromisoformat(date_from))
    if date_to:
        # Include full day: add 1 day so events on date_to day are included
        end = datetime.datetime.fromisoformat(date_to) + datetime.timedelta(days=1)
        q = q.where(CalendarEvent.start < end)
    q = q.order_by(CalendarEvent.start.asc())
    result = await db.execute(q)
    return [_event_out(ev) for ev in result.scalars().all()]
```

#### `POST /calendar/events`

```python
@router.post("/calendar/events", status_code=201)
async def create_event(
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id

    def _parse_dt(s):
        if not s:
            return None
        try:
            return datetime.datetime.fromisoformat(s)
        except ValueError:
            return None

    ev = CalendarEvent(
        workspace_id=ws_id,
        user_id=current_user.id,
        name=body.get("name", "Untitled"),
        start=_parse_dt(body.get("start")),
        stop=_parse_dt(body.get("stop")),
        location=body.get("location") or "",
        description=body.get("description") or "",
        all_day=body.get("all_day", False),
    )
    db.add(ev)
    await db.commit()
    await db.refresh(ev)
    return _event_out(ev)
```

#### `GET /calendar/events/{event_id}`

```python
@router.get("/calendar/events/{event_id}")
async def get_event(
    event_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(CalendarEvent).where(
            CalendarEvent.id == event_id,
            CalendarEvent.workspace_id == ws_id,
        )
    )
    ev = r.scalar_one_or_none()
    if not ev:
        raise HTTPException(404)
    return _event_out(ev)
```

#### `PATCH /calendar/events/{event_id}`

```python
@router.patch("/calendar/events/{event_id}")
async def update_event(
    event_id: int,
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    r = await db.execute(
        select(CalendarEvent).where(
            CalendarEvent.id == event_id,
            CalendarEvent.workspace_id == ws_id,
        )
    )
    ev = r.scalar_one_or_none()
    if not ev:
        raise HTTPException(404)
    for field in ("name", "location", "description", "all_day"):
        if field in body:
            setattr(ev, field, body[field])
    for dt_field in ("start", "stop"):
        if dt_field in body and body[dt_field]:
            try:
                setattr(ev, dt_field, datetime.datetime.fromisoformat(body[dt_field]))
            except ValueError:
                pass
    await db.commit()
    await db.refresh(ev)
    return _event_out(ev)
```

#### `DELETE /calendar/events/{event_id}`

```python

        raise HTTPException(404)
    await db.delete(ev)
    await db.commit()
```
