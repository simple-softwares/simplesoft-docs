# ── GST calculation ────────────────────────────────────────────────────────────

def calc_gst(taxable: float, rate: float, ws_state: str, pos: str) -> dict:
    tax = round(taxable * rate / 100, 2)
    if ws_state == pos:  # intra-state
        half = round(tax / 2, 2)
        return {"cgst": half, "sgst": tax - half, "igst": 0.0}
    return {"cgst": 0.0, "sgst": 0.0, "igst": tax}


# ── Sequence ───────────────────────────────────────────────────────────────────

async def next_invoice_number(workspace_id: int, db: AsyncSession) -> str:
    result = await db.execute(
        select(InvoiceSequence)
        .where(InvoiceSequence.workspace_id == workspace_id)
        .with_for_update()
    )
    seq = result.scalar_one_or_none()
    if not seq:
        seq = InvoiceSequence(workspace_id=workspace_id, last_number=0)
        db.add(seq)
    seq.last_number += 1
    year = datetime.now().year
    return f"INV-{year}-{seq.last_number:05d}"

# ── GST Config helper ──────────────────────────────────────────────────────────

async def get_gst_config(workspace_id: int, db: AsyncSession) -> WorkspaceGSTConfig:
    result = await db.execute(
        select(WorkspaceGSTConfig).where(WorkspaceGSTConfig.workspace_id == workspace_id)
    )
    cfg = result.scalar_one_or_none()
    if not cfg:
        # Return a default if not configured yet
        return WorkspaceGSTConfig(workspace_id=workspace_id, legal_name="My Business", state_code="27")
    return cfg


def build_lines(line_inputs: list[LineIn], ws_state: str, pos: str) -> tuple[list[InvoiceLine], dict]:
    """Build InvoiceLine objects and return invoice totals."""
    lines = []
    subtotal = 0.0
    discount_total = 0.0
    cgst_total = 0.0
    sgst_total = 0.0
    igst_total = 0.0

    for li in line_inputs:
        taxable = round(li.qty * li.unit_price - li.discount_amount, 2)
        gst = calc_gst(taxable, li.gst_rate, ws_state, pos)
        line_total = round(taxable + gst["cgst"] + gst["sgst"] + gst["igst"], 2)

        lines.append(InvoiceLine(
            product_id=li.product_id,
            description=li.description,
            hsn_code=li.hsn_code,
            qty=li.qty,
            unit_price=li.unit_price,
            discount_amount=li.discount_amount,
            gst_rate=li.gst_rate,
            cgst_amount=gst["cgst"],
            sgst_amount=gst["sgst"],
            igst_amount=gst["igst"],
            line_total=line_total,
        ))

        subtotal += round(li.qty * li.unit_price, 2)
        discount_total += li.discount_amount
        cgst_total += gst["cgst"]
        sgst_total += gst["sgst"]
        igst_total += gst["igst"]

    totals = {
        "subtotal": round(subtotal, 2),
        "discount_total": round(discount_total, 2),
        "cgst_total": round(cgst_total, 2),
        "sgst_total": round(sgst_total, 2),
        "igst_total": round(igst_total, 2),
        "total": round(subtotal - discount_total + cgst_total + sgst_total + igst_total, 2),
    }
    return lines, totals

@router.get("", response_model=list[InvoiceOut])
async def list_invoices(
    status: Optional[str] = Query(None),
    customer_id: Optional[int] = Query(None),
    limit: int = Query(50, le=200),
    offset: int = Query(0),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    q = select(Invoice).where(Invoice.workspace_id == current_user.workspace_id)
    if status:
        q = q.where(Invoice.status == status)
    if customer_id:
        q = q.where(Invoice.customer_id == customer_id)
    q = q.order_by(Invoice.created_at.desc()).limit(limit).offset(offset)
    result = await db.execute(q)
    return result.scalars().all()


@router.post("", response_model=InvoiceOut, status_code=201)
async def create_invoice(
    body: InvoiceCreate,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    cfg = await get_gst_config(current_user.workspace_id, db)
    invoice_number = await next_invoice_number(current_user.workspace_id, db)
    lines, totals = build_lines(body.lines, cfg.state_code, body.place_of_supply)

    invoice = Invoice(
        workspace_id=current_user.workspace_id,
        invoice_number=invoice_number,
        customer_id=body.customer_id,
        customer_name=body.customer_name,
        customer_gstin=body.customer_gstin,
        customer_address=body.customer_address,
        invoice_date=body.invoice_date or date.today(),
        due_date=body.due_date,
        place_of_supply=body.place_of_supply,
        notes=body.notes,
        created_by=current_user.id,
        **totals,
    )
    invoice.lines = lines
    db.add(invoice)
    await db.commit()
    await db.refresh(invoice)
    return invoice

@router.patch("/{invoice_id}", response_model=InvoiceOut)
async def update_invoice(
    invoice_id: int,
    body: InvoiceUpdate,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(
        select(Invoice).where(Invoice.id == invoice_id, Invoice.workspace_id == current_user.workspace_id)
    )
    invoice = result.scalar_one_or_none()
    if not invoice:
        raise HTTPException(404, "Invoice not found")
    if invoice.status != "draft":
        raise HTTPException(409, "Invoice cannot be edited after it has been sent. Issue a Credit Note to make corrections.")

    if body.customer_name is not None:
        invoice.customer_name = body.customer_name
    if body.customer_gstin is not None:
        invoice.customer_gstin = body.customer_gstin
    if body.customer_address is not None:
        invoice.customer_address = body.customer_address
    if body.invoice_date is not None:
        invoice.invoice_date = body.invoice_date
    if body.due_date is not None:
        invoice.due_date = body.due_date
    if body.notes is not None:
        invoice.notes = body.notes

    pos = body.place_of_supply or invoice.place_of_supply
    if body.place_of_supply is not None:
        invoice.place_of_supply = body.place_of_supply

    if body.lines is not None:
        # Replace all lines
        for line in invoice.lines:
            await db.delete(line)
        cfg = await get_gst_config(current_user.workspace_id, db)
        new_lines, totals = build_lines(body.lines, cfg.state_code, pos)
        for line in new_lines:
            line.invoice_id = invoice.id
        invoice.lines = new_lines
        for k, v in totals.items():
            setattr(invoice, k, v)

    await db.commit()
    await db.refresh(invoice)
    return invoice
