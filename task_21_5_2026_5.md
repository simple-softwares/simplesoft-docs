```python
def _channel_out(channel: ChatChannel, unread: int = 0) -> dict:
    return {
        "id":                      channel.id,
        "name":                    channel.name,
        "message_unread_counter":  unread,
    }

def _message_out(msg: ChatMessage, user: User | None = None) -> dict:
    return {
        "id":        msg.id,
        "body":      msg.body,
        "date":      msg.created_at.isoformat(),
        "author_id": [user.id, user.full_name] if user else [msg.user_id, "Unknown"],
```

### `GET /chat/channels`

```python
@router.get("/chat/channels")
async def list_channels(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id

    # Auto-seed defaults
    count_r = await db.execute(
        select(func.count(ChatChannel.id)).where(ChatChannel.workspace_id == ws_id)
    )
    if (count_r.scalar() or 0) == 0:
        for name in _DEFAULT_CHANNELS:
            ch = ChatChannel(workspace_id=ws_id, name=name, created_by=current_user.id)
            db.add(ch)
        await db.commit()

    channels_r = await db.execute(
        select(ChatChannel)
        .where(ChatChannel.workspace_id == ws_id)
        .order_by(ChatChannel.created_at.asc())
    )
    channels = channels_r.scalars().all()

    result = []
    for ch in channels:
        # Get last_read_at for this user in this channel
        member_r = await db.execute(
            select(chat_channel_members.c.last_read_at)
            .where(
                chat_channel_members.c.channel_id == ch.id,
                chat_channel_members.c.user_id == current_user.id,
            )
        )
        last_read = member_r.scalar()

        # Count unread messages
        unread_q = select(func.count(ChatMessage.id)).where(ChatMessage.channel_id == ch.id)
        if last_read:
            unread_q = unread_q.where(ChatMessage.created_at > last_read)
        unread_r = await db.execute(unread_q)
        unread = unread_r.scalar() or 0

        result.append(_channel_out(ch, unread))
    return result
```

```python
@router.post("/chat/channels", status_code=201)
async def create_channel(
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    name = (body.get("name") or "").strip()
    if not name:
        raise HTTPException(400, "Channel name required")
    ch = ChatChannel(workspace_id=ws_id, name=name, created_by=current_user.id)
    db.add(ch)
    await db.commit()
    await db.refresh(ch)
    return _channel_out(ch)
```

### `GET /chat/channels/{channel_id}/messages`

Load last 100 messages, oldest first. Load all unique authors in one query.

```python
@router.get("/chat/channels/{channel_id}/messages")
async def list_messages(
    channel_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    # Verify channel belongs to workspace
    ch_r = await db.execute(
        select(ChatChannel).where(ChatChannel.id == channel_id, ChatChannel.workspace_id == ws_id)
    )
    if not ch_r.scalar_one_or_none():
        raise HTTPException(404)

    msgs_r = await db.execute(
        select(ChatMessage)
        .where(ChatMessage.channel_id == channel_id)
        .order_by(ChatMessage.created_at.desc())
        .limit(100)
    )
    msgs = list(reversed(msgs_r.scalars().all()))

    user_ids = {m.user_id for m in msgs}
    users = {}
    if user_ids:
        ur = await db.execute(select(User).where(User.id.in_(user_ids)))
        users = {u.id: u for u in ur.scalars().all()}

    return [_message_out(m, users.get(m.user_id)) for m in msgs]
```

### `POST /chat/channels/{channel_id}/messages`

```python
@router.post("/chat/channels/{channel_id}/messages", status_code=201)
async def send_message(
    channel_id: int,
    body: dict,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    ws_id = current_user.workspace_id
    ch_r = await db.execute(
        select(ChatChannel).where(ChatChannel.id == channel_id, ChatChannel.workspace_id == ws_id)
    )
    if not ch_r.scalar_one_or_none():
        raise HTTPException(404)

    text = (body.get("body") or "").strip()
    if not text:
        raise HTTPException(400, "Message body required")

    msg = ChatMessage(channel_id=channel_id, user_id=current_user.id, body=text)
    db.add(msg)
    await db.commit()
    await db.refresh(msg)
    return _message_out(msg, current_user)
```

### `POST /chat/channels/{channel_id}/mark-read`

Upsert into `chat_channel_members` setting `last_read_at = now()`.

```python
@router.post("/chat/channels/{channel_id}/mark-read")
async def mark_read(
    channel_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    now = datetime.datetime.utcnow()
    # Check if member row exists
    existing_r = await db.execute(
        select(chat_channel_members).where(
            chat_channel_members.c.channel_id == channel_id,
            chat_channel_members.c.user_id == current_user.id,
        )
    )
    if existing_r.first():
        await db.execute(
            update(chat_channel_members)
            .where(
                chat_channel_members.c.channel_id == channel_id,
                chat_channel_members.c.user_id == current_user.id,
            )
            .values(last_read_at=now)
        )
    else:
        await db.execute(
            insert(chat_channel_members).values(
                channel_id=channel_id,
                user_id=current_user.id,
                last_read_at=now,
            )
        )
    await db.commit()
    return {"ok": True}
```
